        // Initializes camera access
        async function setupCamera() {
            try {
                if (videoStream) {
                    videoStream.getTracks().forEach(track => track.stop());
                }
                showMessage("Grant camera access to begin.");
                const stream = await navigator.mediaDevices.getUserMedia({ video: true });
                videoStream = stream;
                video.srcObject = stream;
            } catch (err) {
                showMessage('Error: Camera access was denied. Please allow camera access to use this app.', true);
                console.error("Camera access error: ", err);
                initialOptions.classList.remove('hidden');
                cameraSection.classList.add('hidden');
            }
        }
        
        // Automated photo capture and generation
        async function autoCaptureAndGenerate() {
            if (!videoStream) {
                showMessage("Camera not available. Please allow camera access.", true);
                return;
            }
            
            // Perform the capture
            const imageDataURL = capturePhotoToDataURL();
            
            // Stop video stream and hide camera section
            videoStream.getTracks().forEach(track => track.stop());
            cameraSection.classList.add('hidden');

            const base64Data = imageDataURL.split(',')[1];
            await generateAllStyles(base64Data, 'monthly');
        }

        // Manual photo capture and generation
        async function capturePhoto() {
            const imageDataURL = capturePhotoToDataURL();
            
            // Stop video stream and hide camera section
            videoStream.getTracks().forEach(track => track.stop());
            cameraSection.classList.add('hidden');

            const base64Data = imageDataURL.split(',')[1];
            await generateAllStyles(base64Data, 'standard');
        }

        // Helper function to capture the photo to data URL
        function capturePhotoToDataURL() {
            canvas.width = video.videoWidth;
            canvas.height = video.videoHeight;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            return canvas.toDataURL('image/png');
        }

        async function generateAllStyles(base64Image, generationType) {
            loadingSpinner.classList.remove('hidden');
            stylesGrid.innerHTML = '';
            resultsContainer.classList.add('hidden');
            startOverButtonContainer.classList.add('hidden');
            showMessage("Processing looks...");

            try {
                // Get the JSON list of styles based on user selection
                const gender = genderSelect.value;
                const hairLength = hairLengthSelect.value;
                const styles = await getStyleList(gender, hairLength);
                
                if (!styles || styles.length === 0) {
                    showMessage("Could not get a list of styles. Please try again.", true);
                    return;
                }
                
                // 2. Generate an image for each style, handling failures gracefully
                const imagePromises = styles.map(async style => {
                    try {
                        return await generateImageForStyle(base64Image, style.style_name);
                    } catch (error) {
                        console.error(`Failed to generate image for style "${style.style_name}":`, error);
                        return null; // Return null on failure
                    }
                });

                const generatedImages = await Promise.all(imagePromises);
                
                // 3. Display the results
                styles.forEach((style, index) => {
                    const base64ImageData = generatedImages[index];
                    const styleDiv = document.createElement('div');
                    styleDiv.className = 'bg-gray-50 rounded-xl overflow-hidden shadow-md transition-transform transform hover:scale-105 hover:shadow-lg border border-gray-200';
                    
                    let imageContent;

                    } else {
                        imageContent = `<div class="w-full aspect-square flex flex-col items-center justify-center bg-gray-200 text-gray-500 text-sm p-4 text-center">
                            <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" stroke-width="1.5" stroke="currentColor" class="w-12 h-12 mb-2">
                              <path stroke-linecap="round" stroke-linejoin="round" d="m9.75 9.75 4.5 4.5m0-4.5-4.5 4.5M21 12a9 9 0 1 1-18 0 9 9 0 0 1 18 0Z" />
                            </svg>

                        ${imageContent}
                        <div class="p-4">
                            <h3 class="font-bold text-xl mb-1">${style.style_name}</h3>

        // --- AI Chat Functionality ---
        function openChatModal(styleName) {
            chatTitle.textContent = `Stylist Advice for: ${styleName}`;
            chatModal.style.display = 'flex';
            chatHistory.innerHTML = ''; // Clear previous chat
            chatHistoryData = [{
                role: 'model',
                parts: [{ text: `Hello! I'm your AI stylist. I can give you advice on the "${styleName}" look. What would you like to know?` }]
            }];
            renderChatHistory();
        }

        chatCloseButton.addEventListener('click', () => {
            chatModal.style.display = 'none';
        });

        chatInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter') {
                handleChatSend();
            }
        });

        async function handleChatSend() {
            const userMessage = chatInput.value.trim();
            if (!userMessage) return;

            // Add user message to history
            chatHistoryData.push({ role: 'user', parts: [{ text: userMessage }] });
            renderChatHistory();

            chatInput.value = '';
            chatInput.disabled = true;
            chatSendButton.disabled = true;
            const loadingMessage = addChatMessage('model', '...', false);

            try {
                const response = await getChatResponse(userMessage);
                loadingMessage.remove();
                addChatMessage('model', response, true);
            } catch (error) {
                loadingMessage.remove();
                addChatMessage('model', 'Sorry, I am unable to provide advice at this moment. Please try again.', true);
                console.error('Chat API Error:', error);
            } finally {
                chatInput.disabled = false;
                chatSendButton.disabled = false;
                chatInput.focus();
            }
        }

        function addChatMessage(role, text, isFinal = true) {
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role === 'user' ? 'user-message' : 'ai-message'}`;
            messageDiv.innerHTML = isFinal ? text : `<div class="animate-pulse">...</div>`;
            chatHistory.appendChild(messageDiv);
            chatHistory.scrollTop = chatHistory.scrollHeight;
            return messageDiv;
        }

        function renderChatHistory() {
            chatHistory.innerHTML = '';
            chatHistoryData.forEach(message => {
                if (message.role === 'user') {
                    addChatMessage('user', message.parts[0].text, true);
                } else if (message.role === 'model') {
                    addChatMessage('model', message.parts[0].text, true);
                }
            });
        }

        async function getChatResponse(prompt) {
            const payload = {
                contents: chatHistoryData,
                systemInstruction: {
                    parts: [{ text: "You are a professional AI salon stylist. Provide helpful, concise, and friendly advice on the selected hairstyle. Your responses should be based on general knowledge about hair care, styling, and fashion. Do not offer medical advice." }]
                }
            };
            
            const response = await fetchWithExponentialBackoff(API_ENDPOINT_TEXT, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const result = await response.json();
            return result?.candidates?.[0]?.content?.parts?.[0]?.text;
        }

        // Fetches a list of styles based on gender and hair length
        async function getStyleList(gender, hairLength) {
            let prompt = "";
            const numStyles = 10; // Changed from 15 to 10
            const exclusions = [
                "Curtain Bangs Revival (Mid-Length)", 
                "Buzz Cut (Zero Guard)",
                "Mid-Length Cut" // Excluding this name specifically based on user feedback
            ].join(", ");

            if (gender === 'male') {
                prompt = `Generate a JSON array of ${numStyles} diverse and popular male hairstyle and grooming looks for an Indian customer. The styles should be a mix of short, neat cuts with sharp fades and textured tops, as well as longer, wavy, and layered looks. Exclude the following styles: ${exclusions}. Ensure a variety of tags: "Professional", "Party", "Trendy", and "Casual". Each look should be distinct and popular in contemporary Indian salons. For each style, include: 1. "style_name": a descriptive name, 2. "tag": a single tag from "Professional", "Party", "Trendy", or "Casual", 3. "services_needed": a comma-separated list of services. Ensure the output is a single, valid JSON array.`;
            } else if (gender === 'female') {
                prompt = `Generate a JSON array of ${numStyles} popular Indian female hairstyles. Focus on looks that are ${hairLength} or a mix of ${hairLength} and medium length, inspired by loose waves, braids, and half-up looks suitable for formal or casual wear. Exclude the following styles: ${exclusions}. Ensure a variety of tags: "Professional", "Party", "Trendy", and "Casual". For each style, include: 1. "style_name": a descriptive name, 2. "tag": a single tag from "Professional", "Party", "Trendy", or "Casual", 3. "services_needed": a comma-separated list of services. Ensure the output is a single, valid JSON array.`;
            }

            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: {
                        type: "ARRAY",
                        items: {
                            type: "OBJECT",
                            properties: {
                                "style_name": { "type": "STRING" },
                                "tag": { "type": "STRING", "enum": ["Professional", "Party", "Trendy", "Casual"] },
                                "services_needed": { "type": "STRING" }
                            },
                            "propertyOrdering": ["style_name", "tag", "services_needed"]
                        }
                    }
                }
            };
            return await fetchJsonData(payload);
        }

        async function fetchJsonData(payload) {
            let response = await fetchWithExponentialBackoff(API_ENDPOINT_TEXT, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const result = await response.json();
            const jsonText = result?.candidates?.[0]?.content?.parts?.[0]?.text;
            return jsonText ? JSON.parse(jsonText) : null;
        }

        // Generate a single image with a specific style
        async function generateImageForStyle(base64Image, styleName) {
            const prompt = `Apply the "${styleName}" style to the person's hair and adjust their grooming in this photo. The style should be realistic and fit the person's features.`;
            
            const payload = {
                contents: [{ 
                    parts: [
                        { text: prompt },
                        { inlineData: { mimeType: "image/png", data: base64Image } }
                    ]
                }],
                generationConfig: {
                    responseModalities: ["TEXT", "IMAGE"]
                }
            };
            
            let response = await fetchWithExponentialBackoff(API_ENDPOINT_IMAGE, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const result = await response.json();
            return result?.candidates?.[0]?.content?.parts?.find(p => p.inlineData)?.inlineData?.data;
        }

        // Utility for exponential backoff
        async function fetchWithExponentialBackoff(url, options, retries = 3, delay = 1000) {
            for (let i = 0; i < retries; i++) {
                try {
                    const response = await fetch(url, options);
                    if (response.status === 429) { // Too Many Requests
                        if (i < retries - 1) {
                            console.warn(`Rate limit exceeded. Retrying in ${delay / 1000}s...`);
                            await new Promise(res => setTimeout(res, delay));
                            delay *= 2;
                            continue;
                        }
                    }
                    if (!response.ok) {
                        const errorData = await response.text();
                        throw new Error(`API error: ${response.status} - ${response.statusText} - ${errorData}`);
                    }
                    return response;
                } catch (error) {
                    if (i < retries - 1) {
                        console.warn(`Fetch failed. Retrying in ${delay / 1000}s...`, error);
                        await new Promise(res => setTimeout(res, delay));
                        delay *= 2;
                    } else {
                        throw error;
                    }
                }
            }
        }
