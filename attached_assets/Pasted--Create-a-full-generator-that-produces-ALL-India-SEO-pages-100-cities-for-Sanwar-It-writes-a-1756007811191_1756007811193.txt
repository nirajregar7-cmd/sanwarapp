def read_cities(path):
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    # Deduplicate by slug
    seen = set()
    cleaned = []
    for c in data:
        slug = c.get("slug") or slugify(c["name"])
        if slug in seen: 
            continue
        seen.add(slug)
        cleaned.append({"name": c["name"], "slug": slug, "state": c.get("state", "")})
    return cleaned

def slugify(text):
    t = re.sub(r"[^a-zA-Z0-9]+", "-", text.lower()).strip("-")
    return re.sub(r"-+", "-", t)

def org_schema(brand):
    data = {
      "@context": "https://schema.org",
      "@type": "Organization",
      "name": brand["name"],
      "url": brand["domain"],
      "logo": brand["logo"],
      "contactPoint": [{
        "@type": "ContactPoint",
        "telephone": brand["phone"],
        "contactType": "customer support",
        "areaServed": "IN",
        "availableLanguage": ["en","hi"]
      }]
    }
    return data

def website_search_schema(brand):
    return {
      "@context": "https://schema.org",
      "@type": "WebSite",
      "url": brand["domain"],
      "potentialAction": {
        "@type": "SearchAction",
        "target": f"{brand['domain']}/search?q={{query}}",
        "query-input": "required name=query"
      }
    }

def local_business_schema(service, brand, city=None, near_me=False, india=False):
    data = {
      "@context": "https://schema.org",
      "@type": service["type"],
      "name": f"{brand['name']} – {service['name']}",
      "image": brand["logo"],
      "url": brand["domain"],
      "priceRange": "₹₹",
    }
    if city:
        data["areaServed"] = {"@type":"City", "name": city["name"], "addressCountry":"IN"}
    elif india:
        data["areaServed"] = {"@type":"Country","name":"India"}
    elif near_me:
        data["areaServed"] = {"@type":"Country","name":"India"}
    return data

def breadcrumb_schema(crumbs):
    items = []
    for i,(name,url) in enumerate(crumbs, start=1):
        items.append({"@type":"ListItem","position":i,"name":name,"item":url})
    return {"@context":"https://schema.org","@type":"BreadcrumbList","itemListElement":items}

        ("Are reviews verified?", "Yes. Ratings and reviews are from customers who booked via Sanwar."),
        ("Is there a confirmation fee?", "Some salons may charge a small fee to secure your slot. It’s shown upfront before payment.")

        "@type":"FAQPage",
        "mainEntity":[{"@type":"Question","name":q,"acceptedAnswer":{"@type":"Answer","text":a}} for q,a in qas]

def write_html(path, html):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        f.write(html)
