function findCity(slug: string): CityInfo | undefined {
  return CITIES.find((c) => c.slug === slug);
}

// --- SEO helpers
const BASE_URL = "https://sanwarhub.in"; // change if needed

export async function generateMetadata({ params }: { params: { slug: string } }): Promise<Metadata> {
  const city = findCity(params.slug);
  if (!city) return { title: "City not found | Sanwar" };
  const title = `Book Salons & Barbers in ${city.city} | Sanwar`;
  const description = `Find top salons & barbers in ${city.city}. Book haircuts, grooming, spa and beauty services online with Sanwar – fast, easy and reliable.`;
  const url = `${BASE_URL}/usa/${city.slug}`;
  return {
    title,
    description,
    alternates: { canonical: url },
    openGraph: {
      title,
      description,
      url,
      siteName: "Sanwar",
      type: "website",
      locale: "en_US",
    },
    twitter: {
      card: "summary_large_image",
      title,
      description,
    },
    other: {
      // hreflang examples – expand with real alternates when you add locales
      "hreflang-en-us": url,
      "hreflang-en-in": `${BASE_URL}/in/${city.slug}`,
    },
  };
}
